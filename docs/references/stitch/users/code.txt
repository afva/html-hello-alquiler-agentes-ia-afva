  function toggleRowMenu(event, menuId) {
    event.stopPropagation();
    const target = document.getElementById(menuId);
    const isHidden = target.classList.contains('hidden');
    document.querySelectorAll('.row-action-menu').forEach(function (el) {
      el.classList.add('hidden');
    });
    if (isHidden) {
      target.classList.remove('hidden');
    }
  }

  document.addEventListener('click', function (event) {
    if (!event.target.closest('.row-action-menu')) {
      document.querySelectorAll('.row-action-menu').forEach(function (el) {
        el.classList.add('hidden');
      });
    }
  });

  function filterUsersTable() {
    const query = document.getElementById('user-search-input').value.toLowerCase();
    const plan = document.getElementById('plan-filter').value;
    const status = document.getElementById('status-filter').value;
    document.querySelectorAll('.user-row').forEach(function (row) {
      const name = row.getAttribute('data-name').toLowerCase();
      const email = row.getAttribute('data-email').toLowerCase();
      const rowPlan = row.getAttribute('data-plan');
      const rowStatus = row.getAttribute('data-status');
      const matchesQuery = name.includes(query) || email.includes(query);
      const matchesPlan = plan === 'all' || rowPlan === plan;
      const matchesStatus = status === 'all' || rowStatus === status;
      row.classList.toggle('hidden', !(matchesQuery && matchesPlan && matchesStatus));
    });
  }

    function toggleTheme() {
      const html = document.documentElement;
      if (html.classList.contains('dark')) {
        html.classList.remove('dark');
        localStorage.setItem('agenthub-theme', 'light');
      } else {
        html.classList.add('dark');
        localStorage.setItem('agenthub-theme', 'dark');
      }
      lucide.createIcons();
    }

    // --- SIDEBAR MÓVIL ---
    function toggleMobileSidebar() {
      const sidebar = document.getElementById('sidebar');
      const overlay = document.getElementById('mobile-overlay');
      const isOpen = !sidebar.classList.contains('-translate-x-full');
      if (isOpen) {
        sidebar.classList.add('-translate-x-full');
        overlay.classList.add('hidden');
      } else {
        sidebar.classList.remove('-translate-x-full');
        overlay.classList.remove('hidden');
      }
    }

    function closeMobileSidebar() {
      const sidebar = document.getElementById('sidebar');
      const overlay = document.getElementById('mobile-overlay');
      if (sidebar && overlay) {
        sidebar.classList.add('-translate-x-full');
        overlay.classList.add('hidden');
      }
    }

    function closeAllDropdowns() {

    function toggleDropdown(e, id) {
      e.stopPropagation();
      const menu = document.getElementById(id);
      const isVisible = !menu.classList.contains('hidden');
      closeAllDropdowns();
      if (!isVisible) {
        menu.classList.remove('hidden');
      }
    }

    function toggleAgentSkills(id) {
      const panel = document.getElementById(`agent-skills-panel-${id}`);
      const chevron = document.getElementById(`agent-chevron-${id}`);
      if (panel.classList.contains('hidden')) {
        panel.classList.remove('hidden');
        chevron.classList.add('rotate-180');
      } else {
        panel.classList.add('hidden');
        chevron.classList.remove('rotate-180');
      }
      lucide.createIcons();
    }

    let currentEditingAgent = null;

    function abrirConfigurarAgente(id) {
      const agent = APP_STATE.agents.find(a => a.id === id);
      if (!agent) return;
      currentEditingAgent = agent;

      document.getElementById('modal-agent-title').textContent = `Prompt de Sistema: ${agent.name}`;
      document.getElementById('modal-agent-subtitle').textContent = `Propietario: ${agent.owner}`;
      document.getElementById('modal-agent-prompt').value = agent.prompt;

      openModal('modal-agent-config');
    }

    function guardarPromptAgente() {
      if (currentEditingAgent) {
        currentEditingAgent.prompt = document.getElementById('modal-agent-prompt').value;
        const alert = document.getElementById('prompt-saved-alert');
        alert.classList.remove('hidden');
        setTimeout(() => {
          closeAllModals();
        }, 800);
      }
    }

    function eliminarAgente(id) {
      if (confirm('¿Seguro que deseas eliminar este agente de la flota?')) {
        APP_STATE.agents = APP_STATE.agents.filter(a => a.id !== id);
        renderAgents();
        closeAllDropdowns();
        lucide.createIcons();
      }
    }

            </div>
            <div class="flex justify-between">
              <span class="text-slate-400">Tiempo medio de ejecución:</span>
              <span class="text-slate-800 dark:text-slate-200">140ms</span>

    }

    function eliminarSkill(id) {

    // --- RENDERIZADO: VISTA 6 ERRORES ---
    function renderErrors() {
      const tbody = document.getElementById('errors-table-body');
      if (!tbody) return;

      tbody.innerHTML = APP_STATE.errors.map(err => {
        let badge = '';
        if (err.severity === 'Crítica') {
          badge = '<span class="inline-flex items-center px-2 py-0.5 rounded text-xs font-bold bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300">Crítica</span>';
        } else if (err.severity === 'Alta') {
          badge = '<span class="inline-flex items-center px-2 py-0.5 rounded text-xs font-semibold bg-orange-100 text-orange-700 dark:bg-orange-950 dark:text-orange-300">Alta</span>';
        } else if (err.severity === 'Media') {
          badge = '<span class="inline-flex items-center px-2 py-0.5 rounded text-xs font-medium bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">Media</span>';
        } else {
          badge = '<span class="inline-flex items-center px-2 py-0.5 rounded text-xs font-medium bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300">Baja</span>';
        }

        const isResolvedStyle = err.resolved ? 'opacity-50 line-through' : '';

        return `
          <tr class="hover:bg-slate-50/80 dark:hover:bg-slate-800/40 transition-colors ${isResolvedStyle}">
            <td class="py-4 px-4 sm:px-6 font-mono text-xs text-slate-500">${err.datetime}</td>
            <td class="py-4 px-4 sm:px-6 font-semibold text-slate-900 dark:text-white">${err.agent}</td>
            <td class="py-4 px-4 sm:px-6 text-xs font-medium text-rose-600 dark:text-rose-400">${err.type}</td>
            <td class="py-4 px-4 sm:px-6 text-xs text-slate-600 dark:text-slate-300 max-w-xs truncate">${err.desc}</td>
            <td class="py-4 px-4 sm:px-6">${badge}</td>
            <td class="py-4 px-4 sm:px-6 text-right relative">
              <button onclick="toggleDropdown(event, 'err-dd-${err.id}')" class="dropdown-trigger p-1.5 text-slate-400 hover:text-slate-600 dark:hover:text-slate-200 rounded-lg hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors">
                <i data-lucide="more-vertical" class="w-4 h-4"></i>
              </button>
              <div id="err-dd-${err.id}" class="dropdown-menu hidden absolute right-6 mt-1 w-44 bg-white dark:bg-slate-900 rounded-xl shadow-lg border border-slate-200 dark:border-slate-800 py-1.5 z-40 text-left">
                <button onclick="verDetalleError('${err.id}')" class="w-full px-3.5 py-2 text-xs text-slate-700 dark:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-800 flex items-center gap-2">
                  <i data-lucide="terminal" class="w-3.5 h-3.5 text-brand-600"></i> Ver detalle
                </button>
                <button onclick="marcarResuelto('${err.id}')" class="w-full px-3.5 py-2 text-xs text-emerald-600 dark:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-950/40 flex items-center gap-2">
                  <i data-lucide="check-circle" class="w-3.5 h-3.5"></i> Marcar como resuelto
                </button>
              </div>
            </td>
          </tr>
        `;
      }).join('');
    }

    function verDetalleError(id) {
      const err = APP_STATE.errors.find(e => e.id === id);
      if (!err) return;

      document.getElementById('modal-error-meta').textContent = `${err.agent} • ${err.datetime} • ${err.type}`;
      document.getElementById('modal-error-trace').textContent = err.trace;
      openModal('modal-error-detail');
    }

    function marcarResuelto(id) {
      const err = APP_STATE.errors.find(e => e.id === id);
      if (err) {
        err.resolved = true;
        renderErrors();
        closeAllDropdowns();
        lucide.createIcons();
      }
    }

    function marcarTodosResueltos() {
      APP_STATE.errors.forEach(e => e.resolved = true);
      renderErrors();
      lucide.createIcons();
    }

    document.addEventListener('DOMContentLoaded', function() {
      // Comprobar preferencia de tema almacenada
      const savedTheme = localStorage.getItem('agenthub-theme');
      if (savedTheme === 'dark') {
        document.documentElement.classList.add('dark');
      }

      // Renderizar datos en las vistas
      renderUsers();
      renderAgents();
      renderSkills();
      renderContracts();
      renderErrors();

      // Cargar vista inicial (Dashboard)
      switchView('dashboard');
    });
